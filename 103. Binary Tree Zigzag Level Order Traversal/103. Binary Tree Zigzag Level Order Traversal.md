from collections import deque

class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:
            return []

        level_vals = []
        from_right = False

                        level_nodes.append(node.left)

- 常にappendだと次の階層で入る順番がごちゃごちゃなので崩れる

```python
from collections import deque

class Solution:
    def zigzagLevelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if not root:
            return []
        
        level_nodes = deque([root])
        level_vals = []
        left_to_right = True

        while level_nodes:
            same_level_vals = []
            for _ in range(len(level_nodes)):
                if left_to_right:
                    node = level_nodes.popleft()
                    same_level_vals.append(node.val)

                    if node.left:
                        level_nodes.append(node.left)
                    if node.right:
                        level_nodes.append(node.right)
                
                if not left_to_right:
                    node = level_nodes.pop()
                    same_level_vals.append(node.val)

                    if node.right:
                        level_nodes.appendleft(node.right)
                    if node.left:
                        level_nodes.appendleft(node.left)

            level_vals.append(same_level_vals)
            left_to_right = not left_to_right

        return level_vals
```

- appendleftで新しいものが左に来るようにいれて、same_level_node 内では左に左のノード、右に右のノードが来るようにそろえる
- 常にdequeで木構造を管理するときは左から入れていくことを意識するとミスが防げた
