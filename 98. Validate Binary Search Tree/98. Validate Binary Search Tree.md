# 問題

- 与えられた二分木がBSTかどうか判定する
- BST：左部分木のすべての値 < ノードの値 < 右部分木のすべての値

# Step1

- 左に降りていくときは、今のノードが最大値でそれよりも下のすべての左部分木がそれよりも小さいか
    - 上限を下に伝える
- 右に降りていくときは、今のノードが最小値でそれよりも下のすべての右部分木がそれよりも大きいか
    - 下限を下に伝える
- トップダウンで下に命令と上限か下限を伝えればよい

```python
class Solution:
    def isValidBST(self, root: Optional[TreeNode]) -> bool:
        def valid(node, minimum, maximum):
            if not node:
                return True

            if not (node.val > minimum and node.val < maximum):
                return False

            return valid(node.left, minimum, node.val) and valid(node.right, node.val, maximum)
        
        return valid(root, float("-inf"), float("inf"))
```

```python
class Solution:
    def isValidBST(self, root: Optional[TreeNode]) -> bool:
        
        def dfs(node):
            if not node:
                return True, float('inf'), float('-inf')
            
            left_is_bst, left_min, left_max = dfs(node.left)
            right_is_bst, right_min, right_max = dfs(node.right)
            
            if not left_is_bst or not right_is_bst:
                return False, 0, 0
            
            if not (left_max < node.val < right_min):
                return False, 0, 0

            subtree_min = min(left_min, node.val)
            subtree_max = max(right_max, node.val)
            
            return True, subtree_min, subtree_max
        
        return dfs(root)[0]
```

class Solution:
    def isValidBST(self, root: Optional[TreeNode]) -> bool:
        if not root:
            return True
        
        node_min_max = deque([(root, float('-inf'), float('inf'))])
        
        while node_min_max:
            node, min_val, max_val = node_min_max.popleft()
            
            if not (min_val < node.val < max_val):
                return False
            
            if node.left:
                node_min_max.append((node.left, min_val, node.val))
            
            if node.right:
                node_min_max.append((node.right, node.val, max_val))
        
        return True
