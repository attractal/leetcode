# Step1

- 上から順にトップダウンで足していくのがよさそう
- すべてのノードを走査する必要があるのでBFSもDFSも変わらないだろう
- DFSで書いていく

```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def mergeTrees(self, root1: Optional[TreeNode], root2: Optional[TreeNode]) -> Optional[TreeNode]:
        if (not root1) and (not root2):
            return
        root = TreeNode((root1.val if root1 else 0) + (root2.val if root2 else 0))
        root.left = self.mergeTrees(root1.left if root1 else None, root2.left if root2 else None)
        root.right = self.mergeTrees(root1.right if root1 else None, root2.right if root2 else None) 

        return root
```

- 番兵を使う際は、値を変えない、子を勝手に持たせないなど暗黙の了解も増えるが、valチェックのときに条件分岐の必要がなく短くなる

```python
SENTINEL = TreeNode(0)

class Solution:
    def mergeTrees(self, root1, root2):
        if not root1 and not root2:
            return None

        merged = TreeNode(0)
        stack = [(root1 or SENTINEL, root2 or SENTINEL, merged)]

        while stack:
            n1, n2, m = stack.pop()
            m.val = n1.val + n2.val

            if n1.left is not None or n2.left is not None:
                m.left = TreeNode(0)
                stack.append((
                    n1.left or SENTINEL,
                    n2.left or SENTINEL,
                    m.left
                ))

            if n1.right is not None or n2.right is not None:
                m.right = TreeNode(0)
                stack.append((
                    n1.right or SENTINEL,
                    n2.right or SENTINEL,
                    m.right
                ))

        return merged
```

```python
class Solution:
    def mergeTrees(self, root1, root2):
        if not root1:
            return root2
        if not root2:
            return root1

        root = TreeNode(root1.val + root2.val)
        queue = deque([(root, root1, root2)])

        while queue:
            m, n1, n2 = queue.popleft()

            if n1.left or n2.left:
                if n1.left and n2.left:
                    m.left = TreeNode(n1.left.val + n2.left.val)
                    queue.append((m.left, n1.left, n2.left))
                else:
                    m.left = n1.left or n2.left

            if n1.right or n2.right:
                if n1.right and n2.right:
                    m.right = TreeNode(n1.right.val + n2.right.val)
                    queue.append((m.right, n1.right, n2.right))
                else:
                    m.right = n1.right or n2.right

        return root
```
